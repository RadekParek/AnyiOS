cmake_minimum_required(VERSION 3.22)
project(podnative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ANGLE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/angle")
# AGP packages linked CMake IMPORTED shared libraries. Keep them outside jniLibs
# to avoid packaging the same native library twice.
set(ANGLE_LIB "${ANGLE_ROOT}/lib/${ANDROID_ABI}")
foreach(name EGL GLESv2)
    if(NOT EXISTS "${ANGLE_LIB}/lib${name}_podnative.so")
        message(FATAL_ERROR "Real ANGLE missing. Run scripts/build-angle.sh; no native-GLES substitution is permitted.")
    endif()
    add_library(angle_${name} SHARED IMPORTED)
    set_target_properties(angle_${name} PROPERTIES IMPORTED_LOCATION "${ANGLE_LIB}/lib${name}_podnative.so")
endforeach()
# Same license-reviewed source pins and W^X patch used by the CPU correctness jobs.
set(DYNARMIC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.local/deps/dynarmic")
set(BOOST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.local/boost-headers")
if(NOT EXISTS "${DYNARMIC_ROOT}/CMakeLists.txt" OR NOT EXISTS "${BOOST_ROOT}/boost/version.hpp")
    message(FATAL_ERROR "Pinned CPU sources missing. Run scripts/prepare-cpu.sh.")
endif()
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_FRONTENDS A32 CACHE STRING "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
set(DYNARMIC_ENABLE_NO_EXECUTE_SUPPORT ON CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(Boost_INCLUDE_DIR "${BOOST_ROOT}" CACHE PATH "Pinned Boost headers" FORCE)
set(Boost_NO_BOOST_CMAKE ON CACHE BOOL "" FORCE)
add_subdirectory("${DYNARMIC_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/dynarmic")
target_compile_definitions(dynarmic PRIVATE OAKNUT_USE_WX=1)
add_library(podnative SHARED renderer.cpp bridge.cpp macho.cpp macho_info.cpp macho_container.cpp runtime/memory.cpp runtime/loader.cpp runtime/dyld.cpp runtime/launch.cpp runtime/frameworks.cpp runtime/cpu.cpp runtime/cpu_probe.cpp)
target_compile_options(podnative PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(podnative PRIVATE PODNATIVE_CPU_IN_APP=1)
target_include_directories(podnative PRIVATE "${ANGLE_ROOT}/include")
target_link_libraries(podnative PRIVATE angle_EGL angle_GLESv2 android log vulkan dynarmic)
